// Derived GPU visibility only. CPU terrain heights remain authoritative.
struct Cell {
    direction_height: vec4<f32>,
    corners: array<vec4<f32>,6>,
    metadata: vec4<u32>,
}
struct Params {
    clip_from_world: mat4x4<f32>, camera: vec4<f32>, sun: vec4<f32>, settings: vec4<f32>,
}
struct DrawArgs {
    vertex_count: u32,
    instance_count: atomic<u32>,
    first_vertex: u32,
    first_instance: u32,
}
@group(0) @binding(0) var<uniform> params: Params;
@group(0) @binding(1) var<storage,read> cells: array<Cell>;
@group(0) @binding(2) var<storage,read_write> visible: array<u32>;
@group(0) @binding(3) var<storage,read_write> args: DrawArgs;

// A sphere intersects a clip half-space whenever its center is at least
// -radius * |normal| from the unnormalized plane. The tolerance protects
// shared edges against f32 roundoff at the frustum boundary.
fn outside_plane(plane: vec4<f32>, center: vec3<f32>, radius: f32) -> bool {
    return dot(plane,vec4(center,1.0)) < -(radius+0.25)*length(plane.xyz);
}

fn outside_frustum(center: vec3<f32>, radius: f32) -> bool {
    let m = params.clip_from_world;
    let x = vec4(m[0].x,m[1].x,m[2].x,m[3].x);
    let y = vec4(m[0].y,m[1].y,m[2].y,m[3].y);
    let z = vec4(m[0].z,m[1].z,m[2].z,m[3].z);
    let w = vec4(m[0].w,m[1].w,m[2].w,m[3].w);
    // WebGPU clip depth is 0 <= z <= w, including reverse-Z projections.
    return outside_plane(w+x,center,radius) || outside_plane(w-x,center,radius)
        || outside_plane(w+y,center,radius) || outside_plane(w-y,center,radius)
        || outside_plane(z,center,radius) || outside_plane(w-z,center,radius);
}

@compute @workgroup_size(1)
fn clear_indirect() {
    args.vertex_count = u32(params.settings.w);
    atomicStore(&args.instance_count, 0u);
    args.first_vertex = 0u;
    args.first_instance = 0u;
}

@compute @workgroup_size(128)
fn compact_visible(@builtin(global_invocation_id) id: vec3<u32>) {
    if id.x >= u32(params.settings.y) || id.x >= arrayLength(&cells) { return; }
    // The horizon rejection needs only the 16-byte direction/height field.
    // Fetch corner rays only for cells that survive it.
    let direction_height = cells[id.x].direction_height;
    let radius = params.settings.x;
    let camera_radius = length(params.camera.xyz);
    // The angular horizons of camera and raised terrain overlap. Include a
    // 55m column/foliage margin and base the occluder below the sea surface.
    let occluder = radius - 8.0;
    let height = max(direction_height.w,0.0);
    let top = radius + height + 55.0;
    let angle = acos(clamp(occluder/max(camera_radius,occluder),0.0,1.0))
        + acos(clamp(occluder/top,0.0,1.0)) + 0.018;
    let facing = dot(direction_height.xyz, params.camera.xyz/max(camera_radius,1.0));
    if facing < cos(angle) { return; }

    // Enclose the complete emitted column, not just its cap: exposed skirts
    // descend no lower than sea level, and the tallest decorative tree is
    // 33 * 1.35 = 44.55 m above the cap (55 m retained as a safe allowance).
    // All cap/wall corner rays are unit vectors. Triangle inequality bounds
    // their displacement from the column axis by r * |corner - axis|.
    // A tree's largest horizontal corner is sqrt(2) * 9 * 1.35 < 18 m.
    let degree = min(cells[id.x].metadata.x,6u);
    var maximum_corner_delta_squared = 0.0;
    for (var side=0u; side<degree; side++) {
        let delta = cells[id.x].corners[side].xyz-direction_height.xyz;
        maximum_corner_delta_squared = max(maximum_corner_delta_squared,dot(delta,delta));
    }
    let radial_half_span = (height+55.0)*0.5;
    let center = direction_height.xyz*(radius+radial_half_span);
    let lateral_margin = max((radius+height)*sqrt(maximum_corner_delta_squared),18.0);
    let bound_radius = radial_half_span+lateral_margin;
    if outside_frustum(center,bound_radius) { return; }

    // Every invocation emits at most once, capacity equals authoritative cell
    // count. No partial geometry generation can be published on overflow.
    let slot = atomicAdd(&args.instance_count,1u);
    if slot < arrayLength(&visible) { visible[slot] = id.x; }
}
